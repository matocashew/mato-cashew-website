## តើចំណាត់ថ្នាក់គ្រាប់ស្វាយចន្ទីជាអ្វី?

ការចាត់ថ្នាក់គ្រាប់ស្វាយចន្ទី គឺជាវិធីសម្រាប់ពិពណ៌នា និងបែងចែកគ្រាប់ទៅតាមលក្ខណៈដែលបានកំណត់។ ក្នុងពាណិជ្ជកម្ម ការពិពណ៌នាផលិតផលអាចមានលក្ខណៈច្រើនជាងមួយ។

អ្នកទិញអាចត្រូវកំណត់៖

- **ទម្រង់គ្រាប់**
- **ទំហំគ្រាប់ពេញ** ឬប្រភេទទំហំគ្រាប់បំណែក
- **ថ្នាក់គុណភាព ឬស្ថានភាពគ្រាប់**
- តម្រូវការបច្ចេកទេស ដូចជាសំណើម
- តម្រូវការកែច្នៃ ឬការប្រើប្រាស់ចុងក្រោយ
- តម្រូវការវេចខ្ចប់ និងទីផ្សារ

សម្រាប់ព័ត៌មានលម្អិតអំពីក្របខណ្ឌបច្ចេកទេស និយមន័យ និងស្តង់ដារចំណាត់ថ្នាក់ សូមអាន [ស្តង់ដារចំណាត់ថ្នាក់គ្រាប់ស្វាយចន្ទី](/km/knowledge/cashew-grading-standards/)។

## ចំណាត់ថ្នាក់ ទំហំ និងគុណភាព គឺខុសគ្នា

ការយល់ច្រឡំមួយដែលកើតមានជាញឹកញាប់ គឺការចាត់ទុកលេខទំហំថាជាចំណាត់ថ្នាក់គុណភាពពេញលេញ។

វិធីយល់ដែលត្រឹមត្រូវជាងគឺ៖

ដោយសារពាក្យបច្ចេកទេសមិនដូចគ្នាទាំងស្រុងនៅគ្រប់ស្តង់ដារ ឬលក្ខណៈបច្ចេកទេសរបស់អ្នកទិញទាំងអស់ ស្តង់ដារ ឬឯកសារយោងដែលត្រូវប្រើ គួរត្រូវបានកំណត់ឱ្យច្បាស់។

## គ្រាប់ធំជាង មិនមានន័យថាគុណភាពខ្ពស់ជាងដោយស្វ័យប្រវត្តិ

ទំហំគ្រាប់អាចមានសារៈសំខាន់ក្នុងពាណិជ្ជកម្ម ប៉ុន្តែវាគ្រាន់តែជាលក្ខណៈមួយប៉ុណ្ណោះ។

គ្រាប់ដែលមានទំហំធំជាង មិនមានន័យដោយស្វ័យប្រវត្តិថាមានពណ៌ល្អជាង មានពិការភាពតិចជាង មានសុវត្ថិភាពជាង មានរសជាតិល្អជាង សមស្របសម្រាប់ការកែច្នៃជាង ឬសមស្របសម្រាប់អ្នកទិញគ្រប់ប្រភេទនោះទេ។ ទំហំត្រូវវាយតម្លៃរួមជាមួយគុណភាព និងលក្ខណៈបច្ចេកទេសទាំងមូល។

## ថ្នាក់គុណភាព និងស្ថានភាពគ្រាប់

ការចាត់ថ្នាក់គុណភាពពិចារណាលើលក្ខណៈផ្សេងៗក្រៅពីទំហំ។

អាស្រ័យលើលក្ខណៈបច្ចេកទេសដែលអនុវត្ត លក្ខណៈទាំងនេះអាចរួមមាន៖

- ពណ៌
- ពិការភាព
- ការខូចខាត
- កម្រិតអនុញ្ញាត
- ស្ថានភាព
- លក្ខណៈគុណភាពផ្សេងទៀតដែលបានកំណត់

UNECE DDP-17 ប្រើថ្នាក់គុណភាពដូចជា **Extra, Class I និង Class II**។

AFI ប្រើក្របខណ្ឌពាក្យបច្ចេកទេសខុសគ្នា ដែលរួមមានប្រភេទដូចជា **First Quality Fancy, Second Quality Scorched, Third Quality Special Scorched, Lightly Blemished** និងប្រភេទផ្សេងទៀតដែលបានកំណត់។

សម្រាប់ការពិភាក្សាទូលំទូលាយអំពីពិការភាព ការទទួលយក និងគុណភាពផលិតផល សូមអាន [ស្តង់ដារគុណភាពគ្រាប់ស្វាយចន្ទី](/km/knowledge/cashew-quality-standards/)។

## ស្វែងយល់ពីចំណាត់ថ្នាក់គ្រាប់បំណែក

## ការជ្រើសរើសតាមតម្រូវការរូបរាង

នៅពេលផលិតផលសម្រេចបង្ហាញគ្រាប់ស្វាយចន្ទីឱ្យមើលឃើញច្បាស់ រូបរាងអាចមានឥទ្ធិពលលើការជ្រើសរើស។

អ្នកទិញអាចពិចារណា៖

- តើគ្រាប់ត្រូវរក្សាទម្រង់ពេញឬអត់
- ទំហំគ្រាប់ពេញដែលចង់បាន
- ភាពស៊ីសង្វាក់នៃរូបរាង
- ពណ៌
- ពិការភាពដែលអាចទទួលយកបាន
- ការបាក់បែក
- រូបរាងដែលរំពឹងទុកបន្ទាប់ពីការកែច្នៃបន្ថែម

ដូច្នេះ តម្រូវការគ្រាប់ពេញទំហំធំ គឺខុសពីតម្រូវការគុណភាពខ្ពស់ជាទូទៅ។ ទាំងពីរអាចត្រូវបានកំណត់ក្នុងលក្ខណៈបច្ចេកទេស ប៉ុន្តែមិនគួរយល់ច្រឡំថាជារឿងតែមួយនោះទេ។

## ការជ្រើសរើសតាមតម្រូវការផលិតកម្ម

សម្រាប់ការប្រើប្រាស់ក្នុងផលិតកម្ម ទម្រង់គ្រាប់ដែលត្រូវការអាចមានសារៈសំខាន់ជាងការរក្សាគ្រាប់ទាំងអស់ឱ្យនៅពេញ។

អ្នកកែច្នៃអាចត្រូវការ៖

- គ្រាប់ពុះ ឬគ្រាប់បាក់ដែលមានទម្រង់គ្រប់គ្រង
- ចន្លោះទំហំគ្រាប់បំណែកដែលបានកំណត់
- ទំហំគ្រឿងផ្សំដែលមានភាពស៊ីសង្វាក់
- ឥរិយាបថដែលអាចព្យាករបានក្នុងការកែច្នៃបន្ត

ការប្រើនិយមន័យគ្រាប់បំណែកផ្អែកលើទំហំរន្ធចម្រោះរបស់ AFI បង្ហាញពីមូលហេតុដែលការគ្រប់គ្រងទំហំគ្រាប់បំណែកអាចមានសារៈសំខាន់ចំពោះភាពស៊ីសង្វាក់ក្នុងការផលិត។

ជម្រើសត្រឹមត្រូវនៅតែត្រូវផ្អែកលើដំណើរការ និងលក្ខណៈបច្ចេកទេសដែលអ្នកទិញត្រូវការ ជំនួសឱ្យការប្រើច្បាប់ទូទៅដូចជា "ចំណាត់ថ្នាក់នេះតែងតែសម្រាប់ផលិតផលដុតនំ" ឬ "ចំណាត់ថ្នាក់នោះតែងតែសម្រាប់ផលិតផលបង្អែម"។

## ការជ្រើសរើសតាមតម្រូវការគុណភាព

បន្ទាប់ពីកំណត់ទម្រង់ និងទំហំ អ្នកទិញគួរកំណត់លក្ខណៈគុណភាពដែលមានសារៈសំខាន់។

លក្ខណៈទាំងនេះអាចរួមមាន៖

- ពណ៌ ឬថ្នាក់គុណភាព
- ពិការភាពដែលអាចទទួលយកបាន
- កម្រិតអនុញ្ញាត
- កម្រិតសារធាតុបរទេស នៅពេលពាក់ព័ន្ធ
- សំណើម
- តម្រូវការសុវត្ថិភាពចំណីអាហារ
- លក្ខណៈផ្សេងទៀតដែលបានកំណត់ក្នុងកិច្ចសន្យា

ដូច្នេះ លក្ខណៈបច្ចេកទេសទិញពេញលេញ ត្រូវការព័ត៌មានច្រើនជាង W-number តែមួយ។

## ការពិចារណាអំពីប្រសិទ្ធភាពបន្ទាប់ពីការកែច្នៃ ឬការលីង

រូបរាងគ្រាប់ឆៅអាចមិនបង្ហាញរឿងទាំងមូលទេ នៅពេលអ្នកទិញមានបំណងលីង ឬកែច្នៃគ្រាប់បន្ថែម។

គោលការណ៍ណែនាំអំពី roast test របស់ AFI គាំទ្រការវាយតម្លៃរូបរាងបន្ទាប់ពីការលីង ជាផ្នែកមួយនៃការវាយតម្លៃគុណភាព។ ដូច្នេះ សម្រាប់ការប្រើប្រាស់ដែលប្រសិទ្ធភាពនៃការលីងមានសារៈសំខាន់ អ្នកទិញអាចត្រូវពិចារណា៖

- ពណ៌បន្ទាប់ពីការកែច្នៃ
- ភាពស៊ីសង្វាក់នៃរូបរាង
- ភាពងាយបាក់
- តម្រូវការប្រសិទ្ធភាពផ្សេងទៀតនៃការប្រើប្រាស់ចុងក្រោយ

នេះ **មិនមានន័យថា** ទំហំគ្រាប់ពេញមួយប្រភេទលីងបានល្អជាងមួយប្រភេទទៀតដោយស្វ័យប្រវត្តិនោះទេ។ ប្រសិទ្ធភាពក្នុងការកែច្នៃគួរត្រូវបានវាយតម្លៃជាតម្រូវការដាច់ដោយឡែក។

![ក្របខណ្ឌគំនិតសម្រាប់ការជ្រើសរើសលក្ខណៈបច្ចេកទេសពាណិជ្ជកម្មគ្រាប់ស្វាយចន្ទី ដោយចាប់ផ្តើមពីគោលបំណងប្រើប្រាស់ ទម្រង់គ្រាប់ ទំហំ គុណភាព តម្រូវការបច្ចេកទេស ប្រសិទ្ធភាពកែច្នៃ ការវេចខ្ចប់ និងតម្រូវការរបស់អ្នកទិញ ឬទីផ្សារ។](/images/knowledge/cashew-grade-buyer-selection-flow.jpg)

*ក្របខណ្ឌគំនិតសម្រាប់ការជ្រើសរើសរបស់អ្នកទិញ។ លក្ខណៈបច្ចេកទេសគ្រាប់ស្វាយចន្ទីដែលសមស្រប អាស្រ័យលើគោលបំណងប្រើប្រាស់ ទម្រង់គ្រាប់ ទំហំ គុណភាព តម្រូវការបច្ចេកទេស តម្រូវការកែច្នៃ ការវេចខ្ចប់ និងតម្រូវការរបស់អ្នកទិញ ឬទីផ្សារ។*

លំដាប់ជាក់ស្តែងសម្រាប់ការជ្រើសរើសគឺ៖

1. **កំណត់គោលបំណងប្រើប្រាស់** - កំណត់ការប្រើប្រាស់ចុងក្រោយ និងតម្រូវការរូបរាង។
2. **ជ្រើសរើសទម្រង់គ្រាប់** - គ្រាប់ពេញ គ្រាប់បាក់ គ្រាប់ពុះ ឬគ្រាប់បំណែក។
3. **ជ្រើសរើសតម្រូវការទំហំ** - ចំនួនគ្រាប់ពេញ ឬប្រភេទទំហំគ្រាប់បំណែកដែលបានកំណត់។
4. **កំណត់ថ្នាក់គុណភាព** - ពណ៌ ពិការភាព កម្រិតអនុញ្ញាត និងពាក្យបច្ចេកទេសគុណភាពដែលអនុវត្ត។
5. **កំណត់តម្រូវការបច្ចេកទេស** - សំណើម សុវត្ថិភាពចំណីអាហារ និងលក្ខណៈបច្ចេកទេសផ្សេងទៀតរបស់អ្នកទិញ។
6. **ពិចារណាប្រសិទ្ធភាពក្នុងការកែច្នៃ** - រួមទាំងប្រសិទ្ធភាពបន្ទាប់ពីការលីង ឬការកែច្នៃបន្ថែម នៅពេលពាក់ព័ន្ធ។
7. **ជ្រើសរើសការវេចខ្ចប់** - ទៅតាមតម្រូវការការពារផលិតផល និងពាណិជ្ជកម្ម។
8. **បញ្ជាក់តម្រូវការរបស់អ្នកទិញ និងទីផ្សារ** - រួមទាំងតម្រូវការជាក់លាក់របស់ទីផ្សារគោលដៅ។
9. **ព្រមព្រៀងលើលក្ខណៈបច្ចេកទេសពាណិជ្ជកម្ម** - ធានាថាអ្នកលក់ និងអ្នកទិញប្រើនិយមន័យ និងលក្ខខណ្ឌទទួលយកដូចគ្នា។

ការជ្រើសរើសពាណិជ្ជកម្មគួររួមបញ្ចូលទម្រង់ ទំហំ គុណភាព តម្រូវការបច្ចេកទេស តម្រូវការកែច្នៃ ការវេចខ្ចប់ និងតម្រូវការរបស់អ្នកទិញ។ ទំហំតែមួយមុខមិនគ្រប់គ្រាន់ទេ។

## ចំណាត់ថ្នាក់ + សំណើម + គុណភាព + ការវេចខ្ចប់

ការបញ្ជាទិញពាណិជ្ជកម្មអាចយល់បានថាជាការរួមបញ្ចូលលក្ខណៈបច្ចេកទេសដែលមានទំនាក់ទំនងគ្នា៖

**ទម្រង់គ្រាប់ + ទំហំ + គុណភាព + សំណើម + សុវត្ថិភាពចំណីអាហារ + ការវេចខ្ចប់ + តម្រូវការរបស់អ្នកទិញ**

សំណើមគួរត្រូវបានកំណត់ និងបកស្រាយក្នុងបរិបទនៃតម្រូវការពាណិជ្ជកម្មដែលអនុវត្ត។ សម្រាប់ព័ត៌មានលម្អិតអំពីការវាស់វែង និងស្តង់ដារ សូមអាន [ស្តង់ដារសំណើមគ្រាប់ស្វាយចន្ទី](/km/knowledge/cashew-moisture-standards/)។

ការវេចខ្ចប់ត្រូវការពារស្ថានភាពផលិតផលដែលបានព្រមព្រៀង ក្នុងអំឡុងពេលស្តុក ដឹកជញ្ជូន និងចែកចាយ។ សម្រាប់ប្រព័ន្ធវេចខ្ចប់ និងគោលការណ៍ការពារ សូមអាន [មគ្គុទ្ទេសក៍វេចខ្ចប់គ្រាប់ស្វាយចន្ទី](/km/knowledge/cashew-packaging-guide/)។

## លក្ខណៈបច្ចេកទេសពាណិជ្ជកម្ម និងការនាំចេញ

សម្រាប់ការបញ្ជាទិញអន្តរជាតិ ការជ្រើសរើសផលិតផលត្រូវសមស្របនឹងបរិបទពាណិជ្ជកម្ម និងទីផ្សារគោលដៅផងដែរ។

អ្នកទិញ និងអ្នកលក់អាចត្រូវព្រមព្រៀងលើ៖

- ផលិតផល និងទម្រង់គ្រាប់
- លេខកំណត់ទំហំ
- ថ្នាក់គុណភាព
- តម្រូវការបច្ចេកទេស
- បរិមាណ
- ការវេចខ្ចប់
- ការសម្គាល់លើកញ្ចប់
- ការធ្វើតេស្ត ឬការត្រួតពិនិត្យ នៅពេលពាក់ព័ន្ធ
- ការដឹកជញ្ជូន និងតម្រូវការពាណិជ្ជកម្មផ្សេងទៀត

ការសម្រេចចិត្តអំពីការជ្រើសរើសផលិតផលទាំងនេះ គឺគ្រាន់តែជាផ្នែកមួយនៃការត្រៀមខ្លួនសម្រាប់ការនាំចេញ។ សម្រាប់ក្របខណ្ឌទូលំទូលាយអំពីឯកសារ តម្រូវការទីផ្សារគោលដៅ ភស្តុភារ និងការតាមដានប្រភព សូមអាន [មគ្គុទ្ទេសក៍នាំចេញគ្រាប់ស្វាយចន្ទី](/km/knowledge/cashew-export-guide/)។
